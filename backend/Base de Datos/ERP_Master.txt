-- ============================================================================
-- SISTEMA ERP COMERCIAL Y CONTABLE (NEXOERP / SI2)
-- SCRIPT DDL + DML: BASE DE DATOS CENTRAL SAAS (ERP_MASTER) - POSTGRESQL 16
-- 
-- RESPONSABILIDAD:
--   ADMINISTRA EL ONBOARDING DE CLIENTES (TENANTS), PLANES DE SUSCRIPCIÓN,
--   METADATOS DE CONEXIÓN A LAS BASES DE DATOS DE CADA EMPRESA Y EL REGISTRO
--   HISTÓRICO DE BACKUPS GLOBALES.
-- ============================================================================

-- ============================================================================
-- 1. CATÁLOGOS BASE DEL SAAS
-- ============================================================================

-- 1.1 PLANES DE SUSCRIPCIÓN (MODELO DE NEGOCIO SAAS)
CREATE TABLE PLANES_SUSCRIPCION (
    ID SERIAL PRIMARY KEY,
    CODIGO VARCHAR(30) UNIQUE NOT NULL,
    NOMBRE VARCHAR(100) NOT NULL,
    DESCRIPCION TEXT,
    PRECIO_MENSUAL NUMERIC(12,2) NOT NULL,
    PRECIO_ANUAL NUMERIC(12,2) NOT NULL,
    LIMITE_USUARIOS INTEGER NOT NULL,
    LIMITE_SUCURSALES INTEGER NOT NULL,
    LIMITE_ALMACENES INTEGER NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    ACTIVO BOOLEAN NOT NULL
);

-- 1.2 CIUDAD (CATÁLOGO GEOGRÁFICO NACIONAL)
CREATE TABLE CIUDAD (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(100) NOT NULL,
    PAIS VARCHAR(100) NOT NULL
);


-- ============================================================================
-- 2. REGISTRO DE EMPRESAS CLIENTES (TENANTS)
-- ============================================================================

-- 2.1 EMPRESA (REGISTRO CENTRAL DE TENANTS Y ENRUTAMIENTO A SU PROPIA BD)
CREATE TABLE EMPRESA (
    ID SERIAL PRIMARY KEY,
    RAZON_SOCIAL VARCHAR(200) NOT NULL,
    NOMBRE_COMERCIAL VARCHAR(200) NOT NULL,
    NIT VARCHAR(30) UNIQUE NOT NULL,
    DIRECCION VARCHAR(255),
    TELEFONO VARCHAR(30),
    EMAIL VARCHAR(120),
    MONEDA_PRINCIPAL VARCHAR(10) NOT NULL,
    MONEDA_SECUNDARIA VARCHAR(10),
    LOGO_URL TEXT,
    DB_NAME VARCHAR(100) UNIQUE NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    ACTIVO BOOLEAN NOT NULL
);

-- 2.2 SUSCRIPCIÓN DE EMPRESAS A PLANES SAAS
CREATE TABLE SUSCRIPCION (
    ID SERIAL PRIMARY KEY,
    EMPRESA_ID INTEGER NOT NULL,
    PLAN_ID INTEGER NOT NULL,
    FECHA_INICIO DATE NOT NULL,
    FECHA_FIN DATE NOT NULL,
    MONTO_PAGADO NUMERIC(12,2) NOT NULL,
    ESTADO VARCHAR(30) NOT NULL,
    CONSTRAINT FK_SUSCRIPCION_EMPRESA FOREIGN KEY (EMPRESA_ID) REFERENCES EMPRESA(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE,
    CONSTRAINT FK_SUSCRIPCION_PLAN FOREIGN KEY (PLAN_ID) REFERENCES PLANES_SUSCRIPCION(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);


-- ============================================================================
-- 3. ACCESO GLOBAL Y GESTIÓN DE RESPALDOS
-- ============================================================================

-- 3.1 USUARIOS DE GESTIÓN CENTRAL SAAS (SUPERADMIN Y ACCESO A TENANTS)
CREATE TABLE USUARIO_MASTER (
    ID SERIAL PRIMARY KEY,
    EMPRESA_ID INTEGER,
    EMAIL VARCHAR(100) UNIQUE NOT NULL,
    PASSWORD VARCHAR(255) NOT NULL,
    NOMBRE_COMPLETO VARCHAR(150) NOT NULL,
    ROL_GLOBAL VARCHAR(30) NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    ACTIVO BOOLEAN NOT NULL,
    CONSTRAINT FK_USUARIO_MASTER_EMPRESA FOREIGN KEY (EMPRESA_ID) REFERENCES EMPRESA(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 3.2 REGISTRO CENTRAL DE BACKUPS REALIZADOS POR TENANT
CREATE TABLE HISTORIAL_BACKUPS (
    ID SERIAL PRIMARY KEY,
    EMPRESA_ID INTEGER NOT NULL,
    NOMBRE_ARCHIVO VARCHAR(255) NOT NULL,
    RUTA_STORAGE TEXT NOT NULL,
    TAMANO_BYTES INTEGER NOT NULL,
    ESTADO VARCHAR(30) NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    CONSTRAINT FK_BACKUPS_EMPRESA FOREIGN KEY (EMPRESA_ID) REFERENCES EMPRESA(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);


-- ============================================================================
-- 4. ÍNDICES DE RENDIMIENTO (ERP_MASTER)
-- ============================================================================

CREATE INDEX IDX_MASTER_EMPRESA_NIT ON EMPRESA(NIT);
CREATE INDEX IDX_MASTER_EMPRESA_DB ON EMPRESA(DB_NAME);
CREATE INDEX IDX_MASTER_USUARIO_EMAIL ON USUARIO_MASTER(EMAIL);
CREATE INDEX IDX_MASTER_BACKUPS_EMPRESA ON HISTORIAL_BACKUPS(EMPRESA_ID, FECHA_CREACION DESC);
