
-- ============================================================================
-- MÓDULO 1: CONFIGURACIÓN, SUCURSALES Y SEGURIDAD INTERNA
-- ============================================================================

-- 1.1 CONFIGURACIÓN VISUAL Y DE APARIENCIA DE LA EMPRESA
CREATE TABLE CONFIGURACION_EMPRESA (
    ID SERIAL PRIMARY KEY,
    TEMA VARCHAR(30),
    COLOR_PRIMARIO VARCHAR(20),
    COLOR_SECUNDARIO VARCHAR(20),
    LOGO_PERSONALIZADO_URL TEXT,
    ENCABEZADO_FACTURA TEXT,
    PIE_FACTURA TEXT,
    FECHA_MODIFICACION TIMESTAMP
);

-- 1.2 CIUDAD (CATÁLOGO GEOGRÁFICO DE LA EMPRESA)
CREATE TABLE CIUDAD (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(100) NOT NULL,
    PAIS VARCHAR(100) NOT NULL
);

-- 1.3 SUCURSALES DE LA EMPRESA
CREATE TABLE SUCURSAL (
    ID SERIAL PRIMARY KEY,
    CIUDAD_ID INTEGER NOT NULL,
    NOMBRE VARCHAR(120) NOT NULL,
    DIRECCION VARCHAR(255) NOT NULL,
    TELEFONO VARCHAR(30),
    HORA_APERTURA TIME,
    HORA_CIERRE TIME,
    ACTIVO BOOLEAN NOT NULL,
    CONSTRAINT FK_SUCURSAL_CIUDAD FOREIGN KEY (CIUDAD_ID) REFERENCES CIUDAD(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 1.4 MÓDULOS DEL SISTEMA
CREATE TABLE MODULO (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(100) NOT NULL
);

-- 1.5 FUNCIONES DEL MENÚ Y CASOS DE USO POR MÓDULO
CREATE TABLE FUNCION (
    ID SERIAL PRIMARY KEY,
    MODULO_ID INTEGER NOT NULL,
    NOMBRE VARCHAR(100) NOT NULL,
    CONSTRAINT FK_FUNCION_MODULO FOREIGN KEY (MODULO_ID) REFERENCES MODULO(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 1.6 ROLES DE USUARIO (ADMINISTRADOR, CAJERO, VENDEDOR, ETC.)
CREATE TABLE ROL (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(100) NOT NULL,
    PERMISO VARCHAR(100),
    ACTIVO BOOLEAN NOT NULL
);

-- 1.7 ROL_FUNCION (PERMISOS DE LECTURA / EDICIÓN POR FUNCIÓN)
CREATE TABLE ROL_FUNCION (
    ROL_ID INTEGER NOT NULL,
    FUNCION_ID INTEGER NOT NULL,
    DESCRIPCION VARCHAR(100) NOT NULL,
    PRIMARY KEY (ROL_ID, FUNCION_ID),
    CONSTRAINT FK_ROL_FUNC_ROL FOREIGN KEY (ROL_ID) REFERENCES ROL(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE,
    CONSTRAINT FK_ROL_FUNC_FUNCION FOREIGN KEY (FUNCION_ID) REFERENCES FUNCION(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 1.8 EMPLEADOS DE LA EMPRESA
CREATE TABLE EMPLEADOS (
    ID SERIAL PRIMARY KEY,
    SUCURSAL_ID INTEGER NOT NULL,
    CI VARCHAR(30) UNIQUE NOT NULL,
    NOMBRE VARCHAR(100) NOT NULL,
    APELLIDO VARCHAR(100) NOT NULL,
    CARGO VARCHAR(100) NOT NULL,
    TELEFONO VARCHAR(30),
    EMAIL VARCHAR(100),
    DIRECCION VARCHAR(255),
    FECHA_INGRESO DATE NOT NULL,
    ACTIVO BOOLEAN NOT NULL,
    CONSTRAINT FK_EMPLEADOS_SUCURSAL FOREIGN KEY (SUCURSAL_ID) REFERENCES SUCURSAL(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 1.9 USUARIOS DEL SISTEMA CON ROL DIRECTO
CREATE TABLE USUARIO (
    ID SERIAL PRIMARY KEY,
    ROL_ID INTEGER NOT NULL,
    EMPLEADO_ID INTEGER,
    EMAIL VARCHAR(100) UNIQUE NOT NULL,
    PASSWORD VARCHAR(255) NOT NULL,
    NIVEL_USUARIO VARCHAR(30) NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    ACTIVO BOOLEAN NOT NULL,
    CONSTRAINT FK_USUARIO_ROL FOREIGN KEY (ROL_ID) REFERENCES ROL(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE,
    CONSTRAINT FK_USUARIO_EMPLEADO FOREIGN KEY (EMPLEADO_ID) REFERENCES EMPLEADOS(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 1.10 TOKEN_RECUPERACION DE CONTRASEÑA
CREATE TABLE TOKEN_RECUPERACION (
    ID SERIAL PRIMARY KEY,
    USUARIO_ID INTEGER NOT NULL,
    TOKEN VARCHAR(255) NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    FECHA_EXPIRACION TIMESTAMP NOT NULL,
    ESTADO VARCHAR(20) NOT NULL,
    CONSTRAINT FK_TOKEN_RECUP_USUARIO FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 1.11 NOTIFICACIONES INTERNAS DE LA EMPRESA
CREATE TABLE NOTIFICACIONES (
    ID SERIAL PRIMARY KEY,
    USUARIO_ID INTEGER NOT NULL,
    TITULO VARCHAR(150) NOT NULL,
    MENSAJE TEXT NOT NULL,
    TIPO VARCHAR(30) NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    LEIDO BOOLEAN NOT NULL,
    CONSTRAINT FK_NOTIF_USUARIO FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID)

    ON DELETE CASCADE
);


-- ============================================================================
-- MÓDULO 2: INVENTARIO, ARTÍCULOS Y SUCURSALES
-- ============================================================================

-- 2.1 UNIDADES DE MEDIDA
CREATE TABLE UNIDADES_MEDIDA (
    ID SERIAL PRIMARY KEY,
    CODIGO VARCHAR(20) UNIQUE NOT NULL,
    NOMBRE VARCHAR(100) NOT NULL,
    ABREVIATURA VARCHAR(10) NOT NULL,
    ACTIVO BOOLEAN NOT NULL
);

-- 2.2 CATEGORIAS (ÁRBOL RECURSIVO CON PADRE_ID PARA SUBCATEGORÍAS)
CREATE TABLE CATEGORIAS (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(100) NOT NULL,
    DESCRIPCION TEXT,
    PADRE_ID INTEGER,
    ACTIVO BOOLEAN NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    CONSTRAINT FK_CATEGORIAS_PADRE FOREIGN KEY (PADRE_ID) REFERENCES CATEGORIAS(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 2.3 CATÁLOGO DE PRODUCTOS Y ARTÍCULOS
CREATE TABLE PRODUCTOS (
    ID SERIAL PRIMARY KEY,
    CATEGORIA_ID INTEGER NOT NULL,
    UNIDAD_MEDIDA_ID INTEGER NOT NULL,
    CODIGO VARCHAR(50) UNIQUE NOT NULL,
    NOMBRE VARCHAR(150) NOT NULL,
    DESCRIPCION TEXT,
    PRECIO_VENTA NUMERIC(12,2) NOT NULL,
    COSTO_COMPRA NUMERIC(12,2) NOT NULL,
    STOCK_MINIMO NUMERIC(12,2) NOT NULL,
    STOCK_MAXIMO NUMERIC(12,2),
    IMAGEN_URL TEXT,

    FECHA_CREACION TIMESTAMP NOT NULL,

    ON UPDATE CASCADE
    ON DELETE CASCADE,
    CONSTRAINT FK_PRODUCTOS_UNIDAD_MEDIDA FOREIGN KEY (UNIDAD_MEDIDA_ID) REFERENCES UNIDADES_MEDIDA(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 2.4 INVENTARIO POR SUCURSAL
CREATE TABLE INVENTARIO_SUCURSAL (
    ID SERIAL PRIMARY KEY,
    SUCURSAL_ID INTEGER NOT NULL,
    PRODUCTO_ID INTEGER NOT NULL,
    STOCK_DISPONIBLE NUMERIC(12,2) NOT NULL,
    STOCK_MINIMO NUMERIC(12,2) NOT NULL,
    ULTIMA_ACTUALIZACION TIMESTAMP NOT NULL,
    CONSTRAINT UQ_INV_SUCURSAL_PRODUCTO UNIQUE (SUCURSAL_ID, PRODUCTO_ID),
    CONSTRAINT FK_INV_SUCURSAL_SUCURSAL FOREIGN KEY (SUCURSAL_ID) REFERENCES SUCURSAL(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE,
    CONSTRAINT FK_INV_SUCURSAL_PRODUCTO FOREIGN KEY (PRODUCTO_ID) REFERENCES PRODUCTOS(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
);

-- 2.5 KARDEX VALORADO DE MOVIMIENTOS
CREATE TABLE KARDEX_MOVIMIENTOS (
    ID SERIAL PRIMARY KEY,
    SUCURSAL_ID INTEGER NOT NULL,
    PRODUCTO_ID INTEGER NOT NULL,
    UNIDAD_MEDIDA_ID INTEGER NOT NULL,
    USUARIO_ID INTEGER NOT NULL,
    TIPO_MOVIMIENTO VARCHAR(30) NOT NULL,
    MOTIVO VARCHAR(255),
    DOCUMENTO_REFERENCIA VARCHAR(50),
    CANTIDAD NUMERIC(12,2) NOT NULL,
    COSTO_UNITARIO NUMERIC(12,2) NOT NULL,
    SALDO_CANTIDAD NUMERIC(12,2) NOT NULL,
    SALDO_VALOR NUMERIC(14,2) NOT NULL,
    FECHA_MOVIMIENTO TIMESTAMP NOT NULL,
    CONSTRAINT FK_KARDEX_SUCURSAL FOREIGN KEY (SUCURSAL_ID) REFERENCES SUCURSAL(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE,
    CONSTRAINT FK_KARDEX_PRODUCTO FOREIGN KEY (PRODUCTO_ID) REFERENCES PRODUCTOS(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE,
    CONSTRAINT FK_KARDEX_UNIDAD FOREIGN KEY (UNIDAD_MEDIDA_ID) REFERENCES UNIDADES_MEDIDA(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE,
    CONSTRAINT FK_KARDEX_USUARIO FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID)
    ON UPDATE CASCADE
    ON DELETE CASCADE
